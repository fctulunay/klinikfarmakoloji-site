    c = ''

    if l in FF: c += 'body{font-family:%s,"Liberation Serif",serif!important}' % FF[l]
    return c

os.makedirs(OUT, exist_ok=True)
with sync_playwright() as p:
    br = p.chromium.launch()
    for i, l in enumerate(L):
        t = {k: v[i] for k, v in S.items()}
        for name, fn, w, h in (('infografik', info, 1080, 1080), ('kapak', cover, 1500, 800)):
            pg = br.new_page(viewport={'width': w, 'height': h}); pg.set_content(fn(l, t)); pg.wait_for_timeout(250)
            bad = pg.evaluate(CHECK)
            if bad: print('OVERFLOW', name, l, bad)
            pg.screenshot(path='%s/glp1-benzeri-%s%s.png' % (OUT, name, '' if l == 'tr' else '.' + l)); pg.close()
    br.close()
